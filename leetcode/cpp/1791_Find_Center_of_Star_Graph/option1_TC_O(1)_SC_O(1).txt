class Solution {
public:
    int findCenter(vector<vector<int>>& edges) {
        unordered_set<int> hs;
        for (int i =0; i<edges.size(); i++){
            if ( hs.find(edges[i][0])!=hs.end()){
                return edges[i][0];
            }
            if (hs.find(edges[i][1])!=hs.end()){
                return edges[i][1];
            }
            hs.insert(edges[i][0]);
            hs.insert(edges[i][1]);
        }
        return -1;
    }
};
